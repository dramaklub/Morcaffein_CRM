===== FILE: CRM_Morcaffein.domain/entities/Order.cs =====
[only ApplyPromo and RecalculateTotals changed; the rest of the file is unchanged.
 Replace those two methods with:]

    public void ApplyPromo(Campaign campaign)
    {
        EnsureOrderIsPending();
        if (campaign is null) throw new ArgumentNullException(nameof(campaign));

        decimal discount = campaign.CalculateDiscount(SubTotal);
        var redeem = new CampaignRedeem(campaign.CampaignId, OrderId, discount);
        redeem.AttachCampaign(campaign);   // so RecalculateTotals can re-evaluate later
        CampaignRedeems.Add(redeem);
        RecalculateTotals();
    }

    public void RemovePromo(Campaign campaign)
    {
        if (campaign == null) throw new ArgumentNullException(nameof(campaign));

        EnsureOrderIsPending();

        var redeem = CampaignRedeems.FirstOrDefault(r => r.CampaignId == campaign.CampaignId);
        if (redeem is null)
            return;

        CampaignRedeems.Remove(redeem);
        RecalculateTotals();
    }

    public void RecalculateTotals()
    {
        SubTotal = OrderItems.Sum(item => item.SubTotal);

        // Re-evaluate each applied campaign against the CURRENT SubTotal rather than using
        // the snapshot taken at apply time — otherwise a percentage promo applied to a ₱100
        // cart stays frozen at its ₱10 discount even after the cart grows to ₱200.
        // CampaignRedeem carries the Campaign reference for exactly this reason (EF populates
        // it on load; ApplyPromo attaches it at creation).
        TotalDiscount = CampaignRedeems.Sum(redeem =>
        {
            if (redeem.Campaign is null)
                return redeem.DiscountUsed;   // fall back to snapshot if nav isn't loaded

            var freshDiscount = redeem.Campaign.CalculateDiscount(SubTotal);
            redeem.UpdateDiscountUsed(freshDiscount);
            return freshDiscount;
        });

        TotalAmount = Math.Max(0m, SubTotal - TotalDiscount);
    }
