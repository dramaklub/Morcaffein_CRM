        // GET api/tenant/{companyId}/dashboard/loyalty-tier-progression?months=6
        [HttpGet("loyalty-tier-progression")]
        public async Task<ActionResult<LoyaltyTierProgressionDto>> GetLoyaltyTierProgression(
            int companyId,
            [FromQuery] int months = 6)
        {
            if (months < 1) months = 1;
            if (months > 24) months = 24;

            await using var tenant = await _tenantFactory.CreateAsync(companyId);

            var now = DateTime.UtcNow;
            var result = new LoyaltyTierProgressionDto();

            for (var i = months - 1; i >= 0; i--)
            {
                var monthEnd = new DateTime(now.Year, now.Month, 1, 0, 0, 0, DateTimeKind.Utc)
                    .AddMonths(-i + 1).AddTicks(-1);
                var monthStart = new DateTime(monthEnd.Year, monthEnd.Month, 1, 0, 0, 0, DateTimeKind.Utc);

                var eligibleCustomerIds = await tenant.Customers
                    .AsNoTracking()
                    .Where(c => c.CreatedAt <= monthEnd)
                    .Select(c => c.CustomerId)
                    .ToListAsync();

                var runningTotals = await tenant.PointLedgers
                    .AsNoTracking()
                    .Where(pl => pl.Timestamp <= monthEnd)
                    .GroupBy(pl => pl.CustomerId)
                    .Select(g => new { CustomerId = g.Key, Total = g.Sum(x => x.PointsChange) })
                    .ToDictionaryAsync(x => x.CustomerId, x => x.Total);

                int regular = 0, silver = 0, gold = 0;
                foreach (var customerId in eligibleCustomerIds)
                {
                    var total = runningTotals.TryGetValue(customerId, out var t) ? t : 0;
                    if (total >= GoldThreshold) gold++;
                    else if (total >= SilverThreshold) silver++;
                    else regular++;
                }

                result.Months.Add(new LoyaltyTierMonthPoint
                {
                    MonthLabel = monthStart.ToString("yyyy-MM"),
                    RegularCount = regular,
                    SilverCount = silver,
                    GoldCount = gold,
                });
            }

            return Ok(result);
        }

        // Fixed UTC+8 (Philippines Standard Time — no DST, so a constant offset is
        // correct year-round). Order.OrderDate is stored as DateTime.UtcNow; the
        // heatmap needs to bucket by the hour/day-of-week the sale actually
        // happened in the shop's local time, not raw UTC. If this app ever serves
        // shops outside PH, this needs to become a per-tenant setting instead of a
        // hardcoded constant.
        private static readonly TimeSpan PhilippinesUtcOffset = TimeSpan.FromHours(8);

        // GET api/tenant/{companyId}/dashboard/peak-hours
        // Static transaction velocity map (by design unaffected by range selector)
        [HttpGet("peak-hours")]
        public async Task<ActionResult<PeakHoursDto>> GetPeakHours(int companyId)
        {
            await using var tenant = await _tenantFactory.CreateAsync(companyId);

            var orders = await tenant.Orders
                .AsNoTracking()
                .Where(o => o.Status == "Completed")
                .Select(o => new { o.OrderDate, o.TotalAmount })
                .ToListAsync();

            var cells = orders
                .Select(o => new { LocalDate = o.OrderDate + PhilippinesUtcOffset, o.TotalAmount })
                .GroupBy(o => new { Day = (int)o.LocalDate.DayOfWeek, o.LocalDate.Hour })
                .Select(g => new PeakHoursCell
                {
                    DayOfWeek = g.Key.Day,
                    Hour = g.Key.Hour,
                    TransactionCount = g.Count(),
                    Revenue = g.Sum(x => x.TotalAmount),
                })
                .ToList();

            return Ok(new PeakHoursDto { Cells = cells });
        }

        // ---- Chart 1: Age Demographics vs. Spending Power (Shared) ----

        [HttpGet("age-demographics")]
        public async Task<ActionResult<AgeDemographicsAovDto>> GetAgeDemographics(
            int companyId,
            [FromQuery] int days = 30)
        {
            if (days < 1) days = 1;
            if (days > 365) days = 365;

            var startDate = DateTime.UtcNow.Date.AddDays(-days + 1);

            await using var tenant = await _tenantFactory.CreateAsync(companyId);

            var customers = await tenant.Customers
                .AsNoTracking()
                .Where(c => c.IsActive)
                .Select(c => new
                {
                    c.CustomerId,
                    c.Birthday,
                    TotalSpend = c.Orders.Where(o => o.Status == "Completed" && o.OrderDate >= startDate).Sum(o => (decimal?)o.TotalAmount) ?? 0m,
                    OrderCount = c.Orders.Count(o => o.Status == "Completed" && o.OrderDate >= startDate)
                })
                .ToListAsync();

            var totalMembers = customers.Count;
            var today = DateTime.UtcNow;

            string GetBracket(DateTime? bday)
            {
                if (!bday.HasValue) return "Unknown";
                var age = today.Year - bday.Value.Year;
                if (bday.Value.Date > today.AddYears(-age)) age--;

                if (age < 18) return "<18";
                if (age <= 24) return "18-24";
                if (age <= 34) return "25-34";
                return "35+";
            }

            var bracketsOrder = new[] { "<18", "18-24", "25-34", "35+", "Unknown" };

            var grouped = customers
                .GroupBy(c => GetBracket(c.Birthday))
                .ToDictionary(g => g.Key, g => new
                {
                    Count = g.Count(),
                    TotalSpend = g.Sum(x => x.TotalSpend),
                    TotalOrders = g.Sum(x => x.OrderCount)
                });

            var result = new AgeDemographicsAovDto();

            foreach (var b in bracketsOrder)
            {
                if (grouped.TryGetValue(b, out var data))
                {
                    var aov = data.TotalOrders > 0 ? data.TotalSpend / data.TotalOrders : 0m;
                    var pct = totalMembers > 0 ? (decimal)data.Count / totalMembers * 100m : 0m;

                    result.Brackets.Add(new AgeBracketPoint
                    {
                        BracketLabel = b,
                        ActiveMemberCount = data.Count,
                        PercentageOfTotal = decimal.Round(pct, 1),
                        AverageOrderValue = decimal.Round(aov, 2)
                    });
                }
                else
                {
                    result.Brackets.Add(new AgeBracketPoint
                    {
                        BracketLabel = b,
                        ActiveMemberCount = 0,
                        PercentageOfTotal = 0m,
                        AverageOrderValue = 0m
                    });
                }
            }

            return Ok(result);
        }

        // ---- Chart 2: Active Campaign Sales Lift & Redemptions (Shared) ----

        [HttpGet("campaign-lift")]
        public async Task<ActionResult<ActiveCampaignLiftDto>> GetCampaignLift(
            int companyId,
            [FromQuery] int days = 14)
        {
            if (days < 1) days = 1;
            if (days > 90) days = 90;

            await using var tenant = await _tenantFactory.CreateAsync(companyId);

            var startDate = DateTime.UtcNow.Date.AddDays(-days + 1);

            var orders = await tenant.Orders
                .AsNoTracking()
                .Where(o => o.Status == "Completed" && o.OrderDate >= startDate)
                .Select(o => new { Date = o.OrderDate.Date, o.TotalAmount })
                .ToListAsync();

            var redemptions = await tenant.CampaignRedeems
                .AsNoTracking()
                .Where(cr => cr.RedeemDate >= startDate)
                .Select(cr => cr.RedeemDate.Date)
                .ToListAsync();

            var orderGroup = orders.GroupBy(o => o.Date).ToDictionary(g => g.Key, g => g.Sum(x => x.TotalAmount));
            var redemptionGroup = redemptions.GroupBy(d => d).ToDictionary(g => g.Key, g => g.Count());

            var result = new ActiveCampaignLiftDto();

            for (var d = startDate; d <= DateTime.UtcNow.Date; d = d.AddDays(1))
            {
                var rev = orderGroup.TryGetValue(d, out var r) ? r : 0m;
                var red = redemptionGroup.TryGetValue(d, out var c) ? c : 0;

                result.DailyPoints.Add(new DailyCampaignLiftPoint
                {
                    Date = d,
                    DateLabel = d.ToString("MMM d"),
                    DailyRevenue = rev,
                    DailyRedemptions = red
                });
            }

            return Ok(result);
        }

        [HttpGet("feedback-ratings")]
        public async Task<ActionResult<FeedbackRatingDistributionDto>> GetFeedbackRatings(
            int companyId,
            [FromQuery] int days = 30)
        {
            if (days < 1) days = 1;
            if (days > 365) days = 365;

            var startDate = DateTime.UtcNow.Date.AddDays(-days + 1);

            await using var tenant = await _tenantFactory.CreateAsync(companyId);

            var feedbacks = await tenant.Feedbacks
                .AsNoTracking()
                .Where(f => f.IsActive && f.DateLogged >= startDate)
                .Select(f => f.Rating)
                .ToListAsync();

            var total = feedbacks.Count;
            var avg = total > 0 ? feedbacks.Average() : 0.0;

            var result = new FeedbackRatingDistributionDto
            {
                AverageRating = Math.Round(avg, 1),
                TotalFeedbacks = total
            };

            for (int rating = 1; rating <= 5; rating++)
            {
                var count = feedbacks.Count(r => r == rating);
                var pct = total > 0 ? (decimal)count / total * 100m : 0m;

                result.RatingBreakdown.Add(new RatingDistributionPoint
                {
                    Rating = rating,
                    Count = count,
                    Percentage = decimal.Round(pct, 1)
                });
            }

            return Ok(result);
        }

        [HttpGet("barista-sales")]
        public async Task<ActionResult<BaristaSalesPerformanceDto>> GetBaristaSales(
            int companyId,
            [FromQuery] int days = 30)
        {
            if (!CallerIsManagerOrAdmin())
                return StatusCode(403, new { message = "Only managers and admins can view barista performance." });

            if (days < 1) days = 1;
            if (days > 365) days = 365;

            var startDate = DateTime.UtcNow.Date.AddDays(-days + 1);

            await using var tenant = await _tenantFactory.CreateAsync(companyId);

            var sales = await tenant.Orders
                .AsNoTracking()
                .Where(o => o.Status == "Completed" && o.OrderDate >= startDate)
                .GroupBy(o => o.CreatedBy)
                .Select(g => new
                {
                    EmployeeId = g.Key,
                    OrdersProcessed = g.Count(),
                    TotalRevenue = g.Sum(x => x.TotalAmount)
                })
                .ToListAsync();

            var employeeIds = sales.Select(s => s.EmployeeId).ToList();

            var employees = await tenant.Employees
                .AsNoTracking()
                .Where(e => employeeIds.Contains(e.EmployeeId))
                .ToDictionaryAsync(e => e.EmployeeId, e => (e.FirstName + " " + e.LastName).Trim());

            var result = new BaristaSalesPerformanceDto();

            foreach (var item in sales.OrderByDescending(s => s.TotalRevenue))
            {
                var name = employees.TryGetValue(item.EmployeeId, out var empName) ? empName : $"Staff #{item.EmployeeId}";
                var aov = item.OrdersProcessed > 0 ? item.TotalRevenue / item.OrdersProcessed : 0m;

                result.Baristas.Add(new BaristaPerformancePoint
                {
                    EmployeeId = item.EmployeeId,
                    BaristaName = name,
                    OrdersProcessed = item.OrdersProcessed,
                    TotalRevenue = item.TotalRevenue,
                    AverageOrderValue = decimal.Round(aov, 2)
                });
            }

            return Ok(result);
        }

        [HttpGet("pos-diagnostics")]
        public async Task<ActionResult<PosDiagnosticsDto>> GetPosDiagnostics(
            int companyId,
            [FromQuery] int days = 30)
        {
            if (!CallerIsManagerOrAdmin())
                return StatusCode(403, new { message = "Only managers and admins can view POS diagnostics." });

            if (days < 1) days = 1;
            if (days > 365) days = 365;

            var startDate = DateTime.UtcNow.Date.AddDays(-days + 1);

            await using var tenant = await _tenantFactory.CreateAsync(companyId);

            var completedOrders = await tenant.Orders
                .AsNoTracking()
                .Where(o => o.Status == "Completed" && o.OrderDate >= startDate)
                .ToListAsync();

            var gross = completedOrders.Sum(o => o.SubTotal);
            var campaignDiscounts = completedOrders.Sum(o => o.TotalDiscount);
            var pwdDiscounts = completedOrders.Sum(o => o.PwdSeniorDiscountAmount);
            var net = completedOrders.Sum(o => o.TotalAmount);

            var totalCount = completedOrders.Count;
            var pwdCount = completedOrders.Count(o => o.HasPwdSeniorDiscount);
            var campaignCount = completedOrders.Count(o => o.TotalDiscount > 0);

            return Ok(new PosDiagnosticsDto
            {
                GrossSales = gross,
                TotalCampaignDiscounts = campaignDiscounts,
                TotalPwdSeniorDiscounts = pwdDiscounts,
                NetSales = net,
                TotalTransactions = totalCount,
                PwdSeniorTransactionCount = pwdCount,
                CampaignTransactionCount = campaignCount
            });
        }

        // ---- Chart 8: Campaign ROI & Profit Margin Erosion (Admin Only) ----

        [HttpGet("campaign-roi")]
        public async Task<ActionResult<CampaignRoiDto>> GetCampaignRoi(
            int companyId,
            [FromQuery] int days = 30)
        {
            if (!CallerIsManagerOrAdmin())
                return StatusCode(403, new { message = "Only managers and admins can view campaign ROI reports." });

            if (days < 1) days = 1;
            if (days > 365) days = 365;

            var startDate = DateTime.UtcNow.Date.AddDays(-days + 1);

            await using var tenant = await _tenantFactory.CreateAsync(companyId);

            var campaigns = await tenant.Campaigns
                .AsNoTracking()
                .Include(c => c.CampaignRedeems.Where(cr => cr.RedeemDate >= startDate))
                    .ThenInclude(cr => cr.Order)
                .OrderByDescending(c => c.CampaignId)
                .ToListAsync();

            var result = new CampaignRoiDto();

            foreach (var c in campaigns)
            {
                var redemptions = c.CampaignRedeems.Count;
                var discountCost = c.CampaignRedeems.Sum(cr => cr.DiscountUsed);
                var revenue = c.CampaignRedeems
                    .Where(cr => cr.Order != null && cr.Order.Status == "Completed")
                    .Sum(cr => cr.Order!.TotalAmount);

                result.Campaigns.Add(new CampaignRoiItem
                {
                    CampaignId = c.CampaignId,
                    PromoName = c.PromoName,
                    TotalRedemptions = redemptions,
                    TotalDiscountCost = discountCost,
                    RevenueGenerated = revenue,
                    NetProfitImpact = revenue - discountCost
                });
            }

            return Ok(result);
        }

        // ---- Chart 9: Customer Lifetime Value & Churn Risk (Admin Only) ----

        [HttpGet("customer-ltv-churn")]
        public async Task<ActionResult<CustomerLtvChurnDto>> GetCustomerLtvChurn(
            int companyId,
            [FromQuery] int days = 30)
        {
            if (!CallerIsAdmin())
                return StatusCode(403, new { message = "Only admins can view customer LTV reports." });

            if (days < 1) days = 1;
            if (days > 365) days = 365;

            var startDate = DateTime.UtcNow.Date.AddDays(-days + 1);

            await using var tenant = await _tenantFactory.CreateAsync(companyId);

            var customers = await tenant.Customers
                .AsNoTracking()
                .Where(c => c.IsActive)
                .Select(c => new
                {
                    c.CustomerId,
                    c.PointsBalance,
                    c.LastVisitDate,
                    TotalSpend = c.Orders.Where(o => o.Status == "Completed" && o.OrderDate >= startDate).Sum(o => (decimal?)o.TotalAmount) ?? 0m
                })
                .ToListAsync();

            var now = DateTime.UtcNow;
            var thresholdDays = 14;

            var overallAvg = customers.Count > 0 ? customers.Average(c => c.TotalSpend) : 0m;

            var regular = customers.Where(c => c.PointsBalance < 200).ToList();
            var silver = customers.Where(c => c.PointsBalance >= 200 && c.PointsBalance < 500).ToList();
            var gold = customers.Where(c => c.PointsBalance >= 500).ToList();

            var regAvg = regular.Count > 0 ? regular.Average(c => c.TotalSpend) : 0m;
            var silAvg = silver.Count > 0 ? silver.Average(c => c.TotalSpend) : 0m;
            var goldAvg = gold.Count > 0 ? gold.Average(c => c.TotalSpend) : 0m;

            var atRiskCount = customers.Count(c =>
                c.PointsBalance >= 200 && (now - c.LastVisitDate).TotalDays > thresholdDays);

            return Ok(new CustomerLtvChurnDto
            {
                OverallAverageLtv = decimal.Round(overallAvg, 2),
                RegularTierAvgLtv = decimal.Round(regAvg, 2),
                SilverTierAvgLtv = decimal.Round(silAvg, 2),
                GoldTierAvgLtv = decimal.Round(goldAvg, 2),
                AtRiskHighTierCount = atRiskCount
            });
        }

        [HttpGet("audit-summary")]
        public async Task<ActionResult<AuditLogSummaryDto>> GetAuditSummary(
            int companyId,
            [FromQuery] int days = 30)
        {
            if (!CallerIsAdmin())
                return StatusCode(403, new { message = "Only admins can view audit summary reports." });

            if (days < 1) days = 1;
            if (days > 365) days = 365;

            var startDate = DateTime.UtcNow.Date.AddDays(-days + 1);

            await using var tenant = await _tenantFactory.CreateAsync(companyId);

            var adjustments = await tenant.PointAdjustmentRequests
                .AsNoTracking()
                .Where(a => a.RequestedAt >= startDate)
                .ToListAsync();

            var totalAdjustments = adjustments.Count;
            var approvedAdjustments = adjustments.Count(a => a.Status == "Approved");
            var rejectedAdjustments = adjustments.Count(a => a.Status == "Rejected");

            var feedbackBonusCount = await tenant.PointLedgers
                .AsNoTracking()
                .Where(pl => pl.Type == "FeedbackBonus" && pl.Timestamp >= startDate)
                .CountAsync();

            var pendingCampaigns = await tenant.Campaigns
                .AsNoTracking()
                .Where(c => c.Status == "Pending")
                .CountAsync();

            return Ok(new AuditLogSummaryDto
            {
                TotalManualPointAdjustments = totalAdjustments,
                ApprovedPointAdjustments = approvedAdjustments,
                RejectedPointAdjustments = rejectedAdjustments,
                FeedbackBonusPointsAwarded = feedbackBonusCount,
                PendingCampaignApprovals = pendingCampaigns
            });
        }

        // ---- Recent Orders (Sales per Order KPI Modal) ----

        [HttpGet("recent-orders")]
        public async Task<ActionResult<List<RecentOrderSummaryDto>>> GetRecentOrders(
            int companyId,
            [FromQuery] int days = 30,
            [FromQuery] int limit = 10)
        {
            if (days < 1) days = 1;
            if (days > 365) days = 365;
            if (limit < 1) limit = 10;
            if (limit > 100) limit = 100;

            var startDate = DateTime.UtcNow.Date.AddDays(-days + 1);

            await using var tenant = await _tenantFactory.CreateAsync(companyId);

            var orders = await tenant.Orders
                .AsNoTracking()
                .Include(o => o.Customer)
                .Include(o => o.OrderItems)
                .Where(o => o.Status == "Completed" && o.OrderDate >= startDate)
                .OrderByDescending(o => o.OrderDate)
                .Take(limit)
                .Select(o => new RecentOrderSummaryDto
                {
                    OrderId = o.OrderId,
                    OrderDate = o.OrderDate,
                    CustomerName = o.Customer != null ? (o.Customer.FirstName + " " + o.Customer.LastName).Trim() : "Walk-in Customer",
                    ItemCount = o.OrderItems.Sum(i => i.Quantity),
                    TotalAmount = o.TotalAmount
                })
                .ToListAsync();

            return Ok(orders);
        }

            User.IsInRole("Admin") || User.IsInRole("SuperAdmin");

        private static void custRowStyles(TableLayoutPanel row, float col1, float col2)
        {
            row.ColumnStyles.Clear();
            row.ColumnStyles.Add(new ColumnStyle(SizeType.Percent, col1));
            row.ColumnStyles.Add(new ColumnStyle(SizeType.Percent, col2));
        }

        private static TableLayoutPanel MakeBentoRow(int columns, int height, int bottomGap)
        {
            // No Dock here — these rows are flowed children of _bentoContainer now
            // (a FlowLayoutPanel), and FlowLayoutPanel positions children by flow
            // order using their own Width/Height/Margin, not Dock. Dock on a flowed
            // child fights the flow engine's sizing instead of cooperating with it.
            var row = new TableLayoutPanel
            {
                Height      = height,
                ColumnCount = columns,
                RowCount    = 1,
                Margin      = new Padding(0, 0, 0, bottomGap),
                BackColor   = Color.Transparent,
            };
            for (int i = 0; i < columns; i++)
                row.ColumnStyles.Add(new ColumnStyle(SizeType.Percent, 100f / columns));
            return row;
        }

        private void SyncWidth()
        {
            if (_contentHost == null || _bentoContainer == null) return;
            var w = Math.Max(700,
                _contentHost.ClientSize.Width
                - _contentHost.Padding.Horizontal
                - SystemInformation.VerticalScrollBarWidth);
            _bentoContainer.Width = w;
            foreach (Control row in _bentoContainer.Controls)
                row.Width = w;
        }

        private async Task ReloadAsync()
        {
            try
            {
                var days = _dateFilter.SelectedPreset switch
                {
                    DateRangePreset.Today      => 1,
                    DateRangePreset.SevenDays  => 7,
                    _                          => 30,
                };

                // Loyalty Tier Progression is a trend-over-time chart — it doesn't
                // make sense to tie it to the KPI date filter (Today/7 Days/1 Month
                // would each ask for a useless 1-month or shorter window). Always
                // show a fixed trailing window instead.
                const int loyaltyTierMonths = 6;

                // 1. Static Velocity Heatmap (Not affected by range filter)
                _peakHoursCard.BindData(await _controller.GetPeakHoursAsync());

                // 2. Date-filtered metrics — PosDiagnosticsDto has no standalone chart
                // card in this view anymore, but the KPI pods and Revenue/Sales-per-
                // Order modals still derive their numbers from it.
                _latestPos = await _controller.GetPosDiagnosticsAsync(days);

                _latestAov = _latestPos.TotalTransactions > 0 ? _latestPos.NetSales / _latestPos.TotalTransactions : 0m;
                _revenuePod.UpdateData($"₱{_latestPos.NetSales:N0}");
                _ordersPod.UpdateData($"₱{_latestAov:N2}");
                _promosPod.UpdateData($"{_latestPos.CampaignTransactionCount:N0}");

                _latestCsat = await _controller.GetFeedbackRatingsAsync(days);
                _csatPod.UpdateData(_latestCsat.TotalFeedbacks > 0 ? $"{_latestCsat.AverageRating:0.0} ★" : "—");

                _latestRecentOrders = await _controller.GetRecentOrdersAsync(days, 15);
                // Feeds the Promos Redeemed KPI pod's drill-down modal. The Campaign
                // Lift chart card below is a separate endpoint/DTO (daily revenue vs
                // redemption count), not this one.
                _latestCampaignRoi = await _controller.GetCampaignRoiAsync(days);

                _demographicsCard.BindData(await _controller.GetAgeDemographicsAsync(days));
                _loyaltyTierCard.BindData(await _controller.GetLoyaltyTierProgressionAsync(loyaltyTierMonths));
                _campaignLiftCard.BindData(await _controller.GetCampaignLiftAsync(days));
            }
            catch (Exception ex)
            {
                MessageBox.Show(FindForm(), ex.Message, "Dashboard error");
            }
        }

        private void ShowRevenueModal()
        {
            if (_latestPos == null) return;
            using var modal = new RevenueDetailModal(_latestPos);
            modal.ShowDialog(FindForm());
        }

        private void ShowSalesPerOrderModal()
        {
            using var modal = new SalesPerOrderDetailModal(_latestRecentOrders, _latestAov);
            modal.ShowDialog(FindForm());
        }

        private void ShowPromosModal()
        {
            if (_latestCampaignRoi == null) return;
            using var modal = new PromosRedeemedDetailModal(_latestCampaignRoi);
            modal.ShowDialog(FindForm());
        }

        private void ShowCsatModal()
        {
            if (_latestCsat == null) return;
            using var modal = new CustomerHappinessDetailModal(_latestCsat);
            modal.ShowDialog(FindForm());
        }
    }

        public void SelectPreset(DateRangePreset preset, bool notify = true)
        {
            SelectedPreset = preset;

            var today = DateTime.Today;

            switch (preset)
            {
                case DateRangePreset.Today:
                    StartDate = today;
                    EndDate = today.AddDays(1).AddTicks(-1);
                    _customPickerPanel.Visible = false;
                    break;
                case DateRangePreset.SevenDays:
                    StartDate = today.AddDays(-6);
                    EndDate = today.AddDays(1).AddTicks(-1);
                    _customPickerPanel.Visible = false;
                    break;
                case DateRangePreset.OneMonth:
                    StartDate = today.AddMonths(-1);
                    EndDate = today.AddDays(1).AddTicks(-1);
                    _customPickerPanel.Visible = false;
                    break;
                case DateRangePreset.Custom:
                    StartDate = _dtpStart.Value.Date;
                    EndDate = _dtpEnd.Value.Date.AddDays(1).AddTicks(-1);
                    _customPickerPanel.Visible = true;
                    break;
            }

            UpdateButtonStyles();

            if (notify)
            {
                RangeChanged?.Invoke(this, new DashboardDateRangeEventArgs(SelectedPreset, StartDate, EndDate));
            }
        }

        {
            if (SelectedPreset != DateRangePreset.Custom) return;

            StartDate = _dtpStart.Value.Date;
            EndDate = _dtpEnd.Value.Date.AddDays(1).AddTicks(-1);
            RangeChanged?.Invoke(this, new DashboardDateRangeEventArgs(SelectedPreset, StartDate, EndDate));

                    var count = match?.TransactionCount ?? 0;

                    // Inset cells slightly for a cleaner grid

                        ColHeaderHeight + d * cellH + 1,

                    g.FillRectangle(fill, cellRect);

                    if (count > 0 && cellW > 18 && cellH > 14)
                    {
                        var textColor = count > maxCount * 0.55
                            ? Color.White
                            : BrandPalette.NeutralText;
                        using var textBrush = new SolidBrush(textColor);
                        g.DrawString(count.ToString(), cellFont, textBrush, cellRect,
                            new StringFormat

        private void DrawLegend(Graphics g, int maxCount)
        {
            var y = _gridHost.ClientSize.Height - LegendHeight + 2;
            var legendW = Math.Min(160, _gridHost.ClientSize.Width - RowHeaderWidth - 8);
            var x = RowHeaderWidth;

            using var font = BrandPalette.BrandFontRegular(7f);
            using var brush = new SolidBrush(BrandPalette.NeutralTextMuted);

            g.DrawString("Low", font, brush, x, y);

            var barX = x + 24;
            var barW = legendW - 56;
            const int steps = 8;
            var stepW = barW / steps;

            for (int i = 0; i < steps; i++)
            {
                var intensity = (int)Math.Round(maxCount * (i + 1) / (double)steps);
                using var fill = new SolidBrush(ColorForIntensity(Math.Max(1, intensity), Math.Max(1, maxCount)));
                g.FillRectangle(fill, barX + i * stepW, y + 2, stepW - 1, 10);
            }

            g.DrawString("High", font, brush, barX + barW + 4, y);
        }

        private static string FormatHour(int hour)
        {
            if (hour == 0) return "12a";
            if (hour < 12) return $"{hour}a";
            if (hour == 12) return "12p";
            return $"{hour - 12}p";
        }

        /// <summary>
        /// Empty cells stay a warm neutral; filled cells ramp from Tint90 → Brand.
        /// </summary>
        private static Color ColorForIntensity(int count, int max)
        {
            if (count <= 0) return Color.FromArgb(245, 242, 238); // warm empty
            if (max <= 0) return BrandPalette.Tint75;

            var ratio = Math.Clamp((double)count / max, 0.12, 1.0);
            var brand = BrandPalette.Brand;
            var light = Color.FromArgb(250, 246, 240);

            return Color.FromArgb(
                (int)(light.R + (brand.R - light.R) * ratio),
                (int)(light.G + (brand.G - light.G) * ratio),
                (int)(light.B + (brand.B - light.B) * ratio));
        }

        private void GridHost_MouseMove(object? sender, MouseEventArgs e)
        {
            var (cellW, cellH) = GetCellSize();

            if (e.X < RowHeaderWidth || e.Y < ColHeaderHeight ||
                e.Y > ColHeaderHeight + DayLabels.Length * cellH)
            {
                _cellToolTip.Hide(_gridHost);
                _lastHoverCell = null;
                return;
            }

            var h = (int)((e.X - RowHeaderWidth) / cellW);
            var d = (int)((e.Y - ColHeaderHeight) / cellH);

            if (h < 0 || h >= HourCount || d < 0 || d >= DayLabels.Length)
            {
                _cellToolTip.Hide(_gridHost);
                _lastHoverCell = null;
                return;
            }

            if (_lastHoverCell == (d, h)) return;
            _lastHoverCell = (d, h);

            var hour = StartHour + h;
            var match = _data.Cells.FirstOrDefault(c => c.DayOfWeek == d && c.Hour == hour);
            var count = match?.TransactionCount ?? 0;
            var revenue = match?.Revenue ?? 0m;

            var tip = count > 0
                ? $"{DayLabels[d]} {FormatHour(hour)}  ·  {count} orders  ·  ₱{revenue:N0}"
                : $"{DayLabels[d]} {FormatHour(hour)}  ·  no orders";

            _cellToolTip.Show(tip, _gridHost, e.X + 14, e.Y + 14, 2500);
        }

        private void AddSeries(string name, Color color)
        {
            var series = new Series(name)
            {
                // Column (not StackedColumn) — WinForms Chart automatically
                // clusters same-type Column series side by side per category, so
                // this renders Regular/Silver/Gold as 3 separate bars per month
                // instead of piling them into one stacked bar.
                ChartType = SeriesChartType.Column,
                ChartArea = "main",
                Legend = "main",
                Color = color,
                BorderWidth = 0,
                BorderColor = color,
            };
            // Slightly narrower columns look more modern
            series["PointWidth"] = "0.48";
            _chart.Series.Add(series);
        }

        public void BindData(LoyaltyTierProgressionDto data)
        {
            if (data == null) return;

            foreach (var series in _chart.Series)
                series.Points.Clear();

            if (data.Months == null || data.Months.Count == 0)
            {
                ShowEmpty(true);
                return;
            }

            int totalMembers = 0;
            int nonRegular = 0;

            foreach (var month in data.Months)
            {
                _chart.Series["Regular"].Points.AddXY(FormatMonth(month.MonthLabel), month.RegularCount);
                _chart.Series["Silver"].Points.AddXY(FormatMonth(month.MonthLabel), month.SilverCount);
                _chart.Series["Gold"].Points.AddXY(FormatMonth(month.MonthLabel), month.GoldCount);

                totalMembers = month.RegularCount + month.SilverCount + month.GoldCount;
                nonRegular = month.SilverCount + month.GoldCount;
            }

            // If almost everyone is still Regular, show a soft empty-ish message
            // but still keep the chart visible so the structure is clear
            bool sparse = totalMembers > 0 && nonRegular == 0;

            ShowEmpty(false);

            // Force a clean redraw
            _chart.Invalidate();
        }

        private void ShowEmpty(bool empty)
        {
            _emptyLabel.Visible = empty;
            _chart.Visible = !empty;
        }

        private static string FormatMonth(string monthLabel)
        {
            // "2026-06" → "Jun"
            if (DateTime.TryParse(monthLabel + "-01", out var dt))
                return dt.ToString("MMM");
            return monthLabel;
        }

1. AdminDashboardView.cs — Row 4 (Staff Overrides & Audit Logs,
   AuditLogSummaryControl) is built and data-bound every ReloadAsync
   (`_auditSummaryCard.BindData(audit)` runs) but is NEVER added to

   The card is completely invisible right now. Fix: add
   `_bentoContainer.Controls.Add(row4, 0, 4);` after the row3 line (and bump
   RowCount from 5 if staying on TableLayoutPanel).

2. AdminDashboardView.cs — `var months = Math.Max(1, days / 30); if (days <= 7)
   months = 1;` in ReloadAsync evaluates to exactly 1 for every date-filter
   preset (Today=1day, 7 Days=7days, 1 Month=30days — integer division of any
   of these by 30 rounds to 0 or 1). Loyalty Tier Progression on this dashboard
   only ever shows one month's data. This was already fixed on
   ManagerDashboardView.cs (decoupled from the date filter, always requests a
   fixed 6-month window via GetLoyaltyTierProgressionAsync(6)) — the same fix
   needs to be ported to AdminDashboardView.cs.

3. AdminDashboardView.cs uses a TableLayoutPanel with RowCount=5 but no
   RowStyles ever added to it, combined with every "row" panel being its own
   Dock=Top TableLayoutPanel — this is the exact structural pattern that caused
   ManagerDashboardView's date-filter-renders-above-everything bug (fixed there
   by switching to a plain FlowLayoutPanel(TopDown, WrapContents=false), which
   stacks children in insertion order with no row-styling ambiguity). Admin's
   version hasn't visibly broken yet, but carries the identical fragility and
   should get the same FlowLayoutPanel treatment.

4. Orphaned/unused files — built, compile fine, zero call sites anywhere in the
   app (confirmed via grep for `new ClassName(`):
     - FeedbackRatingDistributionControl.cs — a standalone ratings chart card
       duplicating what CustomerHappinessDetailModal already shows in its modal.
     - BaristaSalesPerformanceControl.cs — Manager dropped it from the view by
       deliberate choice this session; Admin never included it (correctly, it
       was always meant to be Manager-only). Endpoint/DTO still work fine.
     - PosDiagnosticsControl.cs — same situation; note PosDiagnosticsDto ITSELF
       is very much in use (feeds Revenue/Sales-per-Order KPI pods+modals on
       both dashboards), just not as a standalone chart card.

5. CampaignRoiControl.cs, CustomerLtvChurnControl.cs, AuditLogSummaryControl.cs
   — grepped for `SeriesChartType` and got zero hits in all three, meaning they
   likely render as grids/labels rather than actual WinForms.DataVisualization
   charts. Not confirmed by reading them line by line — worth a look if they
   need real charts rather than tables.

6. Fixed tonight, mentioned here so the "before" state is understood if this
