﻿using CRM_Morcaffein.domain.entities;
using Microsoft.AspNetCore.Identity.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore;

namespace CRM_Morcaffein.infrastructure.data;

public class TenantCrmDbContext : IdentityDbContext
{
    // ===== TENANT CRM DOMAIN TABLES =====
    public DbSet<Product> Products => Set<Product>();
    public DbSet<ProductSize> ProductSizes => Set<ProductSize>();
    public DbSet<Customer> Customers => Set<Customer>();
    public DbSet<Employee> Employees => Set<Employee>();
    public DbSet<Credential> Credentials => Set<Credential>();
    public DbSet<Order> Orders => Set<Order>();
    public DbSet<OrderItem> OrderItems => Set<OrderItem>();
    public DbSet<Campaign> Campaigns => Set<Campaign>();
    public DbSet<CampaignRedeem> CampaignRedeems => Set<CampaignRedeem>();
    public DbSet<Feedback> Feedbacks => Set<Feedback>();
    public DbSet<PointLedger> PointLedgers => Set<PointLedger>();
    public DbSet<ProductModifier> ProductModifiers => Set<ProductModifier>();
    public DbSet<OrderItemModifier> OrderItemModifiers => Set<OrderItemModifier>();
    public DbSet<PointAdjustmentRequest> PointAdjustmentRequests => Set<PointAdjustmentRequest>();

    public TenantCrmDbContext(DbContextOptions<TenantCrmDbContext> options) : base(options)
    { }

    protected override void OnConfiguring(DbContextOptionsBuilder optionsBuilder)
    {
        base.OnConfiguring(optionsBuilder);
        optionsBuilder.AddInterceptors(new AuditInterceptor());
    }

    protected override void OnModelCreating(ModelBuilder builder)
    {
        base.OnModelCreating(builder);

        // 1. PRODUCT
        builder.Entity<Product>(entity =>
        {
            entity.HasKey(p => p.ProductId);

            entity.Property(p => p.ProductCode)
                  .HasMaxLength(50)
                  .IsRequired();

            entity.Property(p => p.ProductName)
                  .HasMaxLength(200)
                  .IsRequired();

            entity.Property(p => p.ImageUrl).HasMaxLength(500);
            entity.Property(p => p.LocalImagePath).HasMaxLength(500);

            entity.Property(p => p.BasePrice)
                  .HasPrecision(18, 2);

            entity.HasIndex(p => p.ProductCode)
                  .IsUnique();
        });

        builder.Entity<ProductModifier>(entity =>
        {
            entity.HasKey(pm => pm.ProductModifierId);
            entity.Property(pm => pm.Name).HasMaxLength(100).IsRequired();
            entity.Property(pm => pm.Category).HasMaxLength(50);
            entity.Property(pm => pm.Price).HasPrecision(18, 2);
        });

        builder.Entity<OrderItemModifier>(entity =>
        {
            entity.HasKey(oim => oim.OrderItemModifierId);
            entity.Property(oim => oim.ProductModifierNameSnapshot).HasMaxLength(100);
            entity.Property(oim => oim.UnitPrice).HasPrecision(18, 2);

            entity.HasOne(oim => oim.OrderItem)
                  .WithMany(oi => oi.OrderItemModifiers)
                  .HasForeignKey(oim => oim.OrderItemId)
                  .OnDelete(DeleteBehavior.Cascade);

            entity.HasOne(oim => oim.ProductModifier)
                  .WithMany()
                  .HasForeignKey(oim => oim.ProductModifierId)
                  .OnDelete(DeleteBehavior.Restrict);
        });

        // 2. PRODUCT SIZE
        builder.Entity<ProductSize>(entity =>
        {
            entity.HasKey(ps => ps.ProductSizeId);

            entity.Property(ps => ps.SizeName)
                  .HasMaxLength(50)
                  .IsRequired();

            entity.Property(ps => ps.Price)
                  .HasPrecision(18, 2);

            entity.HasOne(ps => ps.Product)
                  .WithMany(p => p.ProductSizes)
                  .HasForeignKey(ps => ps.ProductId)
                  .OnDelete(DeleteBehavior.Cascade);
        });

        // 3. ORDER
        builder.Entity<Order>(entity =>
        {
            entity.HasKey(o => o.OrderId);
            entity.Property(o => o.SubTotal).HasPrecision(18, 2);
            entity.Property(o => o.TotalDiscount).HasPrecision(18, 2);
            entity.Property(o => o.TotalAmount).HasPrecision(18, 2);
            entity.Property(o => o.PointsEarned).IsRequired();

            entity.HasOne(o => o.Customer)
                  .WithMany(c => c.Orders)
                  .HasForeignKey(o => o.CustomerId)
                  .OnDelete(DeleteBehavior.Restrict);

            entity.HasOne(o => o.Employee)
                  .WithMany()
                  .HasForeignKey(o => o.CreatedBy)
                  .OnDelete(DeleteBehavior.Restrict);
        });

        // 4. ORDER ITEM
        builder.Entity<OrderItem>(entity =>
        {
            entity.HasKey(oi => oi.OrderItemId);

            // Stored Properties
            entity.Property(oi => oi.BaseSizePrice).HasPrecision(18, 2);
            entity.Property(oi => oi.SizeNameSnapshot).HasMaxLength(50);

            // Dynamic Computed Properties (Not stored directly in DB table)
            entity.Ignore(oi => oi.UnitPrice);
            entity.Ignore(oi => oi.SubTotal);

            // Relationships
            entity.HasOne(oi => oi.Order)
                  .WithMany(o => o.OrderItems)
                  .HasForeignKey(oi => oi.OrderId)
                  .OnDelete(DeleteBehavior.Cascade);

            entity.HasOne(oi => oi.Product)
                  .WithMany(p => p.OrderItems)
                  .HasForeignKey(oi => oi.ProductId)
                  .OnDelete(DeleteBehavior.Restrict);

            entity.HasOne(oi => oi.ProductSize)
                  .WithMany()
                  .HasForeignKey(oi => oi.ProductSizeId)
                  .OnDelete(DeleteBehavior.Restrict);
        });

        // 5. CAMPAIGN
        builder.Entity<Campaign>(entity =>
        {
            entity.HasKey(c => c.CampaignId);
            entity.Property(c => c.PromoName).HasMaxLength(200).IsRequired();
            entity.Property(c => c.PromoCategory).HasMaxLength(100).IsRequired();
            entity.Property(c => c.DiscountValue).HasPrecision(18, 2);

            entity.HasOne(c => c.Creator)
                  .WithMany()
                  .HasForeignKey(c => c.MadeBy)
                  .OnDelete(DeleteBehavior.Restrict);

            entity.HasOne(c => c.Approver)
                  .WithMany()
                  .HasForeignKey(c => c.ApprovedBy)
                  .OnDelete(DeleteBehavior.Restrict);
        });

        // 6. CAMPAIGN REDEEM
        builder.Entity<CampaignRedeem>(entity =>
        {
            entity.HasKey(cr => cr.RedeemId);
            entity.Property(cr => cr.DiscountUsed).HasPrecision(18, 2);

            entity.HasOne(cr => cr.Campaign)
                  .WithMany(c => c.CampaignRedeems)
                  .HasForeignKey(cr => cr.CampaignId)
                  .OnDelete(DeleteBehavior.Restrict);

            entity.HasOne(cr => cr.Order)
                  .WithMany(o => o.CampaignRedeems)
                  .HasForeignKey(cr => cr.OrderId)
                  .OnDelete(DeleteBehavior.Cascade);
        });

        // 7. FEEDBACK
        builder.Entity<Feedback>(entity =>
        {
            entity.HasKey(f => f.TicketId);
            entity.Property(f => f.Category).HasMaxLength(100);
            entity.Property(f => f.Status).HasMaxLength(50);
            entity.Property(f => f.DecisionNotes).HasMaxLength(500);

            entity.HasOne(f => f.Customer)
                  .WithMany(c => c.Feedbacks)
                  .HasForeignKey(f => f.CustomerId)
                  .OnDelete(DeleteBehavior.Restrict);

            entity.HasOne(f => f.LoggedByEmployee)
                  .WithMany()
                  .HasForeignKey(f => f.LoggedBy)
                  .OnDelete(DeleteBehavior.Restrict);

            entity.HasOne(f => f.DecidedByEmployee)
                  .WithMany()
                  .HasForeignKey(f => f.DecidedBy)
                  .OnDelete(DeleteBehavior.Restrict);
        });

        // 8. POINT LEDGER
        builder.Entity<PointLedger>(entity =>
        {
            entity.HasKey(pl => pl.PointLedgerId);
            entity.Property(pl => pl.Type).HasMaxLength(50).IsRequired();
            entity.Property(pl => pl.Reason).HasMaxLength(250);

            entity.HasOne(pl => pl.Customer)
                  .WithMany(c => c.PointLedgers)
                  .HasForeignKey(pl => pl.CustomerId)
                  .OnDelete(DeleteBehavior.Cascade);

            entity.HasOne(pl => pl.Order)
                  .WithMany()
                  .HasForeignKey(pl => pl.OrderId)
                  .OnDelete(DeleteBehavior.SetNull);

            entity.HasOne(pl => pl.PerformedByEmployee)
                  .WithMany()
                  .HasForeignKey(pl => pl.PerformedBy)
                  .OnDelete(DeleteBehavior.SetNull);
        });

        // 9. POINT ADJUSTMENT REQUEST
        builder.Entity<PointAdjustmentRequest>(entity =>
        {
            entity.HasKey(r => r.PointAdjustmentRequestId);
            entity.Property(r => r.Reason).HasMaxLength(250).IsRequired();
            entity.Property(r => r.Status).HasMaxLength(50).IsRequired();
            entity.Property(r => r.DecisionNotes).HasMaxLength(250);

            entity.HasOne(r => r.Customer)
                  .WithMany(c => c.PointAdjustmentRequests)
                  .HasForeignKey(r => r.CustomerId)
                  .OnDelete(DeleteBehavior.Cascade);

            entity.HasOne(r => r.Requester)
                  .WithMany()
                  .HasForeignKey(r => r.RequestedBy)
                  .OnDelete(DeleteBehavior.Restrict);

            entity.HasOne(r => r.Approver)
                  .WithMany()
                  .HasForeignKey(r => r.ApprovedBy)
                  .OnDelete(DeleteBehavior.Restrict);
        });
    }
}

using System.Drawing;

namespace CRM_Morcaffein.Theme
{
    /// <summary>
    /// Every color in the UI should come from this class or be a plain white/system neutral.
    /// The brand roasted-coffee brown (#572400) — and tints/tones/shades of it — is the only
    /// hue used for branding, primary actions, and content-area accents (see HeaderBackground,
    /// ButtonPrimaryFill). Large chrome surfaces (the sidebar, in particular) are explicitly
    /// allowed to lean on plain white/light-grey neutrals instead of brand tints for a lighter
    /// feel — see the Nav* tokens below. Status/tier indicators are plain text with no color
    /// coding at all for now (no blues/greens/reds/yellows anywhere).
    /// </summary>
    public static class BrandPalette
    {
        public static readonly Color Brand = ColorTranslator.FromHtml("#572400");

        // --- Shades (brand mixed toward black) : darker, for pressed states / headers ---
        public static readonly Color Shade10 = Shade(Brand, 0.10);
        public static readonly Color Shade25 = Shade(Brand, 0.25);
        public static readonly Color Shade45 = Shade(Brand, 0.45); // pressed button state

        // --- Tints (brand mixed toward white) : lighter, for surfaces / hover / disabled ---
        public static readonly Color Tint10 = Tint(Brand, 0.10);
        public static readonly Color Tint25 = Tint(Brand, 0.25); // hover state
        public static readonly Color Tint50 = Tint(Brand, 0.50);
        public static readonly Color Tint75 = Tint(Brand, 0.75); // subtle card background
        public static readonly Color Tint85 = Tint(Brand, 0.85); // sidebar chrome — a step darker than the page background
        public static readonly Color Tint90 = Tint(Brand, 0.90); // page background
        public static readonly Color Tint95 = Tint(Brand, 0.95); // near-white panel background

        // --- Tones (brand mixed toward mid-grey) : muted, for borders / disabled fills ---
        public static readonly Color Tone35 = Tone(Brand, 0.35);
        public static readonly Color Tone60 = Tone(Brand, 0.60); // disabled control fill
        public static readonly Color Tone80 = Tone(Brand, 0.80); // divider lines

        // --- Neutrals: white/black and brand-mixed greys only ---
        public static readonly Color White = Color.White;
        public static readonly Color Black = Color.Black;
        public static readonly Color NeutralText = Shade(Brand, 0.55);   // body text (brand-tinted near-black)
        public static readonly Color NeutralTextMuted = Tone(Brand, 0.45); // secondary text

        // --- Semantic roles, expressed via VALUE not hue ---
        public static Color SurfacePrimary => White;
        public static Color SurfaceSecondary => Tint95;
        public static Color SurfaceCard => Tint90;
        public static Color Border => Tone80;

        // Table headers: a light, brand-tinted band with dark text — not a full-brand fill.
        // The main color stays reserved for primary actions (buttons) so it doesn't compete
        // for attention across every grid in the app. Swapped with RowSelected below: the
        // header now takes the stronger of the two tints, so it stands out over a selected row.
        public static Color HeaderBackground => Tint50;
        public static Color HeaderText => NeutralText;

        // Selected row highlight — was the header's tint, now the lighter one.
        public static Color RowSelected => Tint90;

        public static Font BrandFontRegular(float size) => TryCreateFont("Bahnschrift SemiLight", size, FontStyle.Regular);
        public static Font BrandFontBold(float size) => TryCreateFont("Bahnschrift SemiBold", size, FontStyle.Bold);

        private static Font TryCreateFont(string family, float size, FontStyle style)
        {
            try
            {
                return new Font(family, size, style);
            }
            catch
            {
                // Falls back gracefully if the custom font isn't installed on a given machine.
                return new Font(FontFamily.GenericSansSerif, size, style);
            }
        }

        private static Color Tint(Color c, double amount) => Mix(c, Color.White, amount);
        private static Color Shade(Color c, double amount) => Mix(c, Color.Black, amount);

        // "Tone" = mix toward a neutral mid-grey derived from the brand itself, for muted/disabled elements.
        private static Color Tone(Color c, double amount)
        {
            var grey = Mix(c, Color.Black, 0.5);
            grey = Mix(grey, Color.White, 0.5);
            return Mix(c, grey, amount);
        }

        private static Color Mix(Color a, Color b, double amount)
        {
            amount = Math.Clamp(amount, 0.0, 1.0);
            int r = (int)Math.Round(a.R + (b.R - a.R) * amount);
            int g = (int)Math.Round(a.G + (b.G - a.G) * amount);
            int bl = (int)Math.Round(a.B + (b.B - a.B) * amount);
            return Color.FromArgb(255, r, g, bl);
        }
    }

namespace CRM_Morcaffein.Theme

    /// <summary>
    /// FlatStyle.Flat buttons don't repaint their fill when Enabled flips false — they just
    /// grey the text and keep whatever BackColor was set, so a disabled primary button still
    /// looks like a brand-colored, seemingly-clickable button. Call TrackEnabledState once
    /// (after the button's "active" colors are set) on any button whose Enabled is toggled at
    /// runtime, and it'll swap to the neutral disabled palette automatically.
    /// </summary>
    public static class ButtonStyler
    {
        public static void TrackEnabledState(Button button)
        {
            var activeBackColor = button.BackColor;
            var activeForeColor = button.ForeColor;

            void Apply()
            {
                button.BackColor = button.Enabled ? activeBackColor : BrandPalette.ButtonDisabledFill;
                button.ForeColor = button.Enabled ? activeForeColor : BrandPalette.ButtonDisabledText;
            }

            button.EnabledChanged += (s, e) => Apply();
            Apply(); // the button may already be disabled at hook-up time (e.g. designer default)
        }
    }
}

        public bool CanRequestAdjustment => IsManager(EmployeeSession.Current?.Employee);

        /// <summary>Tier is a business rule (point thresholds), not stored data — computed here, not on Customer.</summary>
        public static string GetTier(int pointsBalance) => pointsBalance switch
        {
            >= 500 => "Gold",
            >= 200 => "Silver",
            _ => "Regular",
        };

        public static bool IsManager(Employee? employee) =>
            employee != null && (
                employee.Role.Equals("Manager", StringComparison.OrdinalIgnoreCase) ||
                employee.Role.Equals("Admin", StringComparison.OrdinalIgnoreCase) ||
                employee.Role.Equals("SuperAdmin", StringComparison.OrdinalIgnoreCase));

        public static bool IsAdmin(Employee? employee) =>
            employee != null && (
                employee.Role.Equals("Admin", StringComparison.OrdinalIgnoreCase) ||
                employee.Role.Equals("SuperAdmin", StringComparison.OrdinalIgnoreCase));

        public Customer Register(string firstName, string lastName, string phone, string? email, DateTime? birthday)
        {
            if (string.IsNullOrWhiteSpace(firstName))
                throw new ArgumentException("First name is required.");
            if (string.IsNullOrWhiteSpace(lastName))
                throw new ArgumentException("Last name is required.");

            var cleanedPhone = new string(phone.Where(char.IsDigit).ToArray());
            if (cleanedPhone.Length < 10)
                throw new ArgumentException("Contact number must have at least 10 digits.");

            var cleanedEmail = string.Empty;
            if (!string.IsNullOrWhiteSpace(email))
            {
                if (!email.Contains('@') || !email.Contains('.'))
                    throw new ArgumentException("Invalid email format.");
                cleanedEmail = email.Trim().ToLower();
            }

            var customer = new Customer
            {
                FirstName = firstName.Trim(),
                LastName = lastName.Trim(),
                ContactNumber = cleanedPhone,
                Email = cleanedEmail,
                Birthday = birthday,
                LastVisitDate = DateTime.UtcNow,
            };

        public void RequestAdjustment(int customerId, int amount, string reason, int requestedByEmployeeId)
        {
            if (!CanRequestAdjustment)
                throw new InvalidOperationException("Only managers (or above) can request a point adjustment.");
            if (amount == 0)
                throw new ArgumentException("Points change cannot be zero.");
            if (string.IsNullOrWhiteSpace(reason))
                throw new ArgumentException("A reason is required.");

            var request = new PointAdjustmentRequest
            {
                CustomerId = customerId,
                PointsChange = amount,
                Reason = reason.Trim(),
                RequestedBy = requestedByEmployeeId,

            _pointAdjustmentService.Save(request);

        public void ApproveAdjustment(int requestId, int approvedByEmployeeId)
        {
            if (!CanApprove)
                throw new InvalidOperationException("Only admins can approve point adjustment requests.");

            var request = _pointAdjustmentService.GetById(requestId);
            if (!request.Status.Equals("Pending", StringComparison.OrdinalIgnoreCase))
                throw new InvalidOperationException($"This request has already been {request.Status}.");
            if (approvedByEmployeeId == request.RequestedBy)
                throw new InvalidOperationException("An employee cannot approve their own adjustment request.");

            var customer = _customerService.GetById(request.CustomerId);
            var newBalance = customer.PointsBalance + request.PointsChange;
            if (newBalance < 0)
                throw new InvalidOperationException("Adjustment would result in a negative points balance.");

            customer.PointsBalance = newBalance;
            customer.LastVisitDate = DateTime.UtcNow;
            _customerService.AddLedgerEntry(new PointLedger
            {
                CustomerId = customer.CustomerId,
                PointsChange = request.PointsChange,
                Type = "Adjustment",
                Reason = request.Reason,
                PerformedBy = approvedByEmployeeId,
            });

            request.Status = "Approved";
            request.ApprovedBy = approvedByEmployeeId;
            request.DecidedAt = DateTime.UtcNow;
            _pointAdjustmentService.Save(request);
        }

                throw new InvalidOperationException($"This request has already been {request.Status}.");

            request.Status = "Rejected";
            request.ApprovedBy = decidedByEmployeeId;
            request.DecidedAt = DateTime.UtcNow;
            request.DecisionNotes = notes.Trim();
            _pointAdjustmentService.Save(request);
        }

using CRM_Morcaffein.domain.entities;
using CRM_Morcaffein.Services;

namespace CRM_Morcaffein.Controllers
{
    /// <summary>
    /// Controller for the Transactions/POS tab (MVC). One instance represents a single
    /// in-progress sale for one customer, created when the View's customer search finds
    /// someone. Owns every business rule for the cart/promo/checkout flow — cart-item
    /// pricing, discount calculation, promo eligibility, points-earned math — since Order/
    /// OrderItem/Campaign/CampaignRedeem are plain data and IOrderService/ICampaignService/
    /// ICustomerService are pure persistence with no rules of their own.
    ///
    /// Order/Customer are exposed directly (read-only) — the View reads Model state off the
    /// Controller after each call, same as any MVC Controller exposing the Model it just
    /// updated.
    /// </summary>
    public class TransactionController
    {
        private readonly IOrderService _orderService;
        private readonly IProductCatalogService _catalogService;
        private readonly ICampaignService _campaignService;
        private readonly ICustomerService _customerService;

        public Order Order { get; private set; }
        public Customer Customer { get; private set; }

        public IReadOnlyList<string> Categories { get; }

        public decimal SubTotal => Order.SubTotal;
        public decimal TotalDiscount => Order.TotalDiscount;
        public decimal TotalAmount => Order.TotalAmount;
        public bool HasItems => Order.OrderItems.Count > 0;
        public IReadOnlyList<OrderItem> CartItems => Order.OrderItems.ToList();

        public TransactionController(
            IOrderService orderService,
            IProductCatalogService catalogService,
            ICampaignService campaignService,
            ICustomerService customerService,
            Customer customer,
            int createdByEmployeeId)
        {
            _orderService = orderService;
            _catalogService = catalogService;
            _campaignService = campaignService;
            _customerService = customerService;
            Customer = customer;
            Order = _orderService.StartOrder(customer.CustomerId, createdByEmployeeId);
            Categories = _catalogService.GetCategories();
        }

        /// <summary>
        /// Re-points the SAME in-progress order at a different customer (Quick Register
        /// relink flow). Not used by the main POS flow — search-and-Go there starts a fresh
        /// Controller/Order instead (see TransactionView.SelectCustomer).
        /// </summary>
        public void SwitchCustomer(Customer newCustomer)
        {
            Order = _orderService.LinkCustomerToOrder(Order, newCustomer);
            Customer = newCustomer;
        }

        public IReadOnlyList<Product> GetProducts(string category, string searchText)
        {
            IEnumerable<Product> products = string.IsNullOrEmpty(category)
                ? _catalogService.GetProducts()
                : _catalogService.GetByCategory(category);

            if (!string.IsNullOrWhiteSpace(searchText))
                products = products.Where(p => p.ProductName.Contains(searchText, StringComparison.OrdinalIgnoreCase));

            return products.ToList();
        }

        public IReadOnlyList<ProductModifier> GetModifiers() => _catalogService.GetModifiers();

        public void AddItemToCart(Product product, ProductSize size, int quantity, IEnumerable<ProductModifier> modifiers)
        {
            EnsurePending();
            if (quantity <= 0)
                throw new ArgumentException("Quantity must be greater than zero.");

            var existing = Order.OrderItems.FirstOrDefault(i => i.ProductSizeId == size.ProductSizeId);
            if (existing != null)
            {
                existing.Quantity += quantity;
                RecalculateItemPricing(existing);
            }
            else
            {
                var item = _orderService.CreateOrderItem(Order, product, size, quantity);
                foreach (var modifier in modifiers)
                    _orderService.AddModifierToItem(item, modifier);
                RecalculateItemPricing(item);
            }

            RecalculateOrderTotals();
        }

        public void UpdateCartItem(OrderItem item, ProductSize newSize, int newQuantity, IEnumerable<ProductModifier> newModifiers)
        {
            EnsurePending();
            if (newQuantity <= 0)
                throw new ArgumentException("Quantity must be greater than zero.");

            if (item.ProductSizeId == newSize.ProductSizeId)
            {
                item.Quantity = newQuantity;
                item.OrderItemModifiers.Clear();
                foreach (var modifier in newModifiers)
                    _orderService.AddModifierToItem(item, modifier);
                RecalculateItemPricing(item);
                RecalculateOrderTotals();
            }
            else
            {
                // No "change size in place" operation — drop the old line, add a fresh one.
                _orderService.RemoveOrderItem(Order, item);
                AddItemToCart(item.Product!, newSize, newQuantity, newModifiers); // recalculates totals itself
            }
        }

        public void RemoveItem(OrderItem item)
        {
            _orderService.RemoveOrderItem(Order, item);
            RecalculateOrderTotals();
        }

        public IReadOnlyList<Campaign> GetEligiblePromos() =>
            _campaignService.GetActiveCampaigns().Where(c => CanRedeem(c, Customer, Order)).ToList();

        public void ApplyPromo(Campaign campaign)
        {
            EnsurePending();
            var discount = CalculateDiscount(campaign, Order.SubTotal);
            _orderService.CreateCampaignRedeem(Order, campaign, discount);
            RecalculateOrderTotals();
        }

        public void RemovePromo(Campaign campaign)
        {
            var redeem = Order.CampaignRedeems.FirstOrDefault(r => r.CampaignId == campaign.CampaignId);
            if (redeem is null) return;

            _orderService.RemoveCampaignRedeem(Order, redeem);
            RecalculateOrderTotals();
        }

        /// <summary>Clears every applied promo, looking each one up by ID so it works even if a campaign has since become ineligible.</summary>
        public void ClearPromos()
        {
            foreach (var redeem in Order.CampaignRedeems.ToList())
                _orderService.RemoveCampaignRedeem(Order, redeem);

            RecalculateOrderTotals();
        }

        public Order CompleteSale()
        {
            EnsurePending();
            if (Order.OrderItems.Count == 0)
                throw new InvalidOperationException("Order cannot be empty.");

            Order.Status = "Completed";
            Order.PointsEarned = (int)(Order.TotalAmount / 10m); // 1 point per ₱10 spent, net of discounts

            if (Order.PointsEarned > 0)
            {
                Customer.PointsBalance += Order.PointsEarned;
                Customer.LastVisitDate = DateTime.UtcNow;
                _customerService.AddLedgerEntry(new PointLedger
                {
                    CustomerId = Customer.CustomerId,
                    PointsChange = Order.PointsEarned,
                    Type = "Earned",
                    Reason = $"Order #{Order.OrderId} Spend",
                    OrderId = Order.OrderId,
                });
            }

            _orderService.CompleteOrder(Order);
            return Order;
        }

        // =====================================================================
        // BUSINESS RULES (previously lived on Order/Campaign/CampaignRedeem —
        // moved here since the domain entities are plain data now)
        // =====================================================================

        private void EnsurePending()
        {
            if (!Order.Status.Equals("Pending", StringComparison.OrdinalIgnoreCase))
                throw new InvalidOperationException($"Order cannot be modified in '{Order.Status}' state.");
        }

        private static void RecalculateItemPricing(OrderItem item)
        {
            item.UnitPrice = item.BaseSizePrice + item.OrderItemModifiers.Sum(m => m.UnitPrice);
            item.SubTotal = item.Quantity * item.UnitPrice;
        }

        /// <summary>
        /// Re-evaluates each applied campaign against the CURRENT SubTotal rather than a
        /// frozen snapshot — a percentage promo applied to a ₱100 cart shouldn't stay
        /// frozen at its ₱10 discount once the cart grows to ₱200.
        /// </summary>
        private void RecalculateOrderTotals()
        {
            Order.SubTotal = Order.OrderItems.Sum(i => i.SubTotal);

            decimal totalDiscount = 0m;
            foreach (var redeem in Order.CampaignRedeems)
            {
                var campaign = redeem.Campaign ?? _campaignService.GetById(redeem.CampaignId);
                var discount = campaign != null ? CalculateDiscount(campaign, Order.SubTotal) : redeem.DiscountUsed;
                redeem.DiscountUsed = discount;
                totalDiscount += discount;
            }
            Order.TotalDiscount = totalDiscount;

            Order.TotalAmount = Math.Max(0m, Order.SubTotal - Order.TotalDiscount);
        }

        private static bool IsCurrentlyActive(Campaign campaign)
        {
            var now = DateTime.UtcNow;
            return campaign.IsActive && now >= campaign.StartDate && now <= campaign.EndDate;
        }

        private static bool CanRedeem(Campaign campaign, Customer customer, Order order)
        {
            if (!IsCurrentlyActive(campaign)) return false;

            if (campaign.MinPurchaseAmount.HasValue && order.SubTotal < campaign.MinPurchaseAmount.Value)
                return false;

            if (!campaign.TargetTier.Equals("All", StringComparison.OrdinalIgnoreCase) &&
                !CustomerController.GetTier(customer.PointsBalance).Equals(campaign.TargetTier, StringComparison.OrdinalIgnoreCase))
                return false;

            if (campaign.PromoCategory.Equals("Birthday", StringComparison.OrdinalIgnoreCase) && !IsBirthdayToday(customer))
                return false;

            if (campaign.PromoCategory.Equals("Inactive", StringComparison.OrdinalIgnoreCase) && !IsInactive(customer))
                return false;

            var previousRedemptions = campaign.CampaignRedeems.Count(r => r.Order != null && r.Order.CustomerId == customer.CustomerId);
            return previousRedemptions < campaign.MaxRedemptionsPerCustomer;
        }

        private static bool IsBirthdayToday(Customer customer) =>
            customer.Birthday?.Month == DateTime.UtcNow.Month && customer.Birthday?.Day == DateTime.UtcNow.Day;

        private static bool IsInactive(Customer customer) => (DateTime.UtcNow - customer.LastVisitDate).Days > 20;

        private static decimal CalculateDiscount(Campaign campaign, decimal orderTotal)
        {
            if (orderTotal <= 0) return 0m;
            if (campaign.MinPurchaseAmount.HasValue && orderTotal < campaign.MinPurchaseAmount.Value) return 0m;

            decimal discount;
            if (campaign.DiscountType.Equals("Percentage", StringComparison.OrdinalIgnoreCase))
            {
                discount = orderTotal * (campaign.DiscountValue / 100m);
                if (campaign.MaxDiscountAmount.HasValue && discount > campaign.MaxDiscountAmount.Value)
                    discount = campaign.MaxDiscountAmount.Value;
            }
            else
            {
                discount = campaign.DiscountValue;
            }

            return discount > orderTotal ? orderTotal : discount;
        }
    }
}

namespace CRM_Morcaffein.Services.Mock
{
    public class MockCustomerService : ICustomerService
    {
        private readonly MockDataStore _store;

        public MockCustomerService(MockDataStore store) => _store = store;

        public Customer? FindByPhoneOrId(string query)
        {
            query = query.Trim();
            if (int.TryParse(query, out var id))
            {
                var byId = _store.Customers.FirstOrDefault(c => c.CustomerId == id);
                if (byId != null) return byId;
            }

            var digitsOnly = new string(query.Where(char.IsDigit).ToArray());
            if (digitsOnly.Length == 0) return null;

            return _store.Customers.FirstOrDefault(c => c.ContactNumber.Contains(digitsOnly));
        }

        public IReadOnlyList<Customer> Search(string query)
        {
            if (string.IsNullOrWhiteSpace(query))
                return _store.Customers;

            query = query.Trim();
            return _store.Customers.Where(c =>
                c.FullName.Contains(query, StringComparison.OrdinalIgnoreCase) ||
                c.ContactNumber.Contains(query) ||
                c.CustomerId.ToString() == query
            ).ToList();
        }

        public Customer GetById(int customerId) =>
            _store.Customers.First(c => c.CustomerId == customerId);

        public Customer Save(Customer customer)
        {
            if (customer.CustomerId == 0)
            {
                customer.CustomerId = _store.NextCustomerId();
                _store.Customers.Add(customer);
            }
            return customer;
        }

        public IReadOnlyList<PointLedger> GetLedger(int customerId) =>
            GetById(customerId).PointLedgers.OrderByDescending(l => l.Timestamp).ToList();

        public IReadOnlyList<Order> GetOrderHistory(int customerId) =>
            _store.Orders.Where(o => o.CustomerId == customerId).OrderByDescending(o => o.OrderDate).ToList();

        public IReadOnlyList<Customer> GetAll() => _store.Customers;

        public void AddLedgerEntry(PointLedger entry)
        {
            entry.PointLedgerId = _store.NextPointLedgerId();
            var customer = _store.Customers.First(c => c.CustomerId == entry.CustomerId);
            entry.Customer = customer;
            customer.PointLedgers.Add(entry);
        }

            Font = BrandPalette.BrandFontBold(9.5f),

        private void BuildSizeOptions()
        {
            foreach (var size in _sizes)
            {
                var radio = new RadioButton
                {
                    Text = $"{size.SizeName} — ₱{size.Price:0.##}",
                    Dock = DockStyle.Top,
                    Height = 26,
                    Checked = size.ProductSizeId == SelectedSize.ProductSizeId,
                    Font = BrandPalette.BrandFontRegular(9.5f),
                };
                radio.CheckedChanged += (s, e) =>
                {
                    if (radio.Checked)
                    {
                        SelectedSize = size;
                        RefreshPriceLabel();
                    }
                };
                _sizePanel.Controls.Add(radio);
            }
            // Panel.Controls.Add with Dock=Top stacks in reverse visual order (last added is
            // topmost) — reverse the child order so sizes read top-to-bottom as seeded.
            var children = _sizePanel.Controls.Cast<Control>().ToList();
            _sizePanel.Controls.Clear();
            for (int i = children.Count - 1; i >= 0; i--)
                _sizePanel.Controls.Add(children[i]);
        }

        private void BuildModifierOptions(IReadOnlyList<ProductModifier> allModifiers, OrderItem? editingItem)
        {
            foreach (var modifier in allModifiers)
            {
                var isPreSelected = editingItem != null &&
                    editingItem.OrderItemModifiers.Any(m => m.ProductModifierId == modifier.ProductModifierId);

                var check = new CheckBox
                {
                    Text = $"{modifier.Name} (+₱{modifier.Price:0.##})",
                    Dock = DockStyle.Top,
                    Height = 26,
                    Checked = isPreSelected,
                    Font = BrandPalette.BrandFontRegular(9.5f),
                };
                check.CheckedChanged += (s, e) => RefreshPriceLabel();
                _modifierChecks[modifier] = check;
                _modifiersPanel.Controls.Add(check);
            }
            var children = _modifiersPanel.Controls.Cast<Control>().ToList();
            _modifiersPanel.Controls.Clear();
            for (int i = children.Count - 1; i >= 0; i--)
                _modifiersPanel.Controls.Add(children[i]);
        }

        private Panel BuildQuantityStepper()
        {
            var row = new Panel { Dock = DockStyle.Top, Height = 40 };

            var minusBtn = MakeStepperButton("-");
            minusBtn.Left = 0;
            minusBtn.Click += (s, e) =>
            {
                Quantity = Math.Max(1, Quantity - 1);
                _qtyValueLabel.Text = Quantity.ToString();
                RefreshPriceLabel();
            };

            var plusBtn = MakeStepperButton("+");
            plusBtn.Left = minusBtn.Width + 44;
            plusBtn.Click += (s, e) =>
            {
                Quantity += 1;
                _qtyValueLabel.Text = Quantity.ToString();
                RefreshPriceLabel();
            };

            _qtyValueLabel.Left = minusBtn.Width + 4;
            _qtyValueLabel.Top = 4;
            _qtyValueLabel.Height = 32;
            _qtyValueLabel.Text = Quantity.ToString();

            row.Controls.Add(minusBtn);
            row.Controls.Add(_qtyValueLabel);
            row.Controls.Add(plusBtn);
            return row;
        }

        private static Button MakeStepperButton(string text) => new()
        {
            Text = text,
            Width = 32,
            Height = 32,
            Top = 4,
            FlatStyle = FlatStyle.Flat,
            BackColor = BrandPalette.ButtonSecondaryFill,
            ForeColor = BrandPalette.ButtonSecondaryText,
            Font = BrandPalette.BrandFontBold(12f),
        };

        private void RefreshPriceLabel()
        {
            _priceLabel.Text = $"Total: ₱{CalculatedLineTotal:0.00}";
            _priceLabel.Font = BrandPalette.BrandFontBold(13f);
            _priceLabel.ForeColor = BrandPalette.Brand;
        }
    }

            var initials = GetInitials(EmployeeSession.Current?.DisplayName);
            using var font = BrandPalette.BrandFontBold(11f);
            var textSize = g.MeasureString(initials, font);

            var parts = fullName.Split(' ', StringSplitOptions.RemoveEmptyEntries);

            {
                0 => "?",
                1 => parts[0][..1].ToUpperInvariant(),
                _ => (parts[0][..1] + parts[^1][..1]).ToUpperInvariant(),
            };
        }

    /// <summary>
    /// One row in the Order box: "Drink Name (Size) +Mods … ₱SubTotal [- 2x +] ✕" — a compact
    /// inline quantity stepper instead of a data-grid row, matching a modern POS's order list
    /// rather than the boxed-table convention the rest of the app uses.
    ///
    /// Absolute-positioned children recomputed on every OnResize. Deliberately NOT Dock-based
    /// (a Dock.Fill name label at full row height wraps before AutoEllipsis kicks in) and NOT
    /// TableLayoutPanel (absolute columns get clipped when the host is narrower than expected).
    ///
    /// The ✕ is a Label, not a Button — WinForms FlatStyle buttons enforce a minimum width
    /// larger than the 28px we want, so a Button ✕ visually overlapped the + stepper button.
    /// Labels have no such floor.
    /// </summary>

    {
        public OrderItem Item { get; }

        protected override void OnHandleCreated(EventArgs e)
        {

        private void LayoutChildren()
        {
            // OnResize fires from the base Panel constructor (via the Height assignment at the
            // top of our own constructor) BEFORE the child fields are assigned. Bail until
            // everything's wired up — real layout runs from OnHandleCreated and every resize.
            if (_nameLabel is null || _priceLabel is null || _rightCluster is null ||
                _stepper is null || _removeLabel is null)
                return;

            var width = ClientSize.Width;
            if (width <= 0)

using CRM_Morcaffein.domain.entities;
using CRM_Morcaffein.Theme;

namespace CRM_Morcaffein.winforms.Views.Controls
{
    /// <summary>
    /// A single drink card: image placeholder + name + "from ₱price", with a row of size
    /// quick-add buttons pinned to the bottom (one per Product.ProductSizes — tapping one
    /// adds that size, qty 1, no modifiers, straight to the cart). Clicking the card body
    /// itself (not a size button) opens the full customizer modal instead, for picking
    /// modifiers or a quantity other than 1. Image loading (LocalImagePath / ImageUrl /
    /// placeholder) is deliberately left as a simple placeholder glyph for this pass — swap
    /// the placeholder paint for a real Image draw once product photography exists.
    /// </summary>
    public class ProductCardControl : Panel
    {
        public Product Product { get; }

        /// <summary>Card body clicked (not a size button) — host opens the full customizer modal.</summary>
        public event EventHandler? CardClicked;

        /// <summary>A size button was tapped — host adds that size directly to the cart, qty 1, no modifiers.</summary>
        public event EventHandler<ProductSize>? SizeQuickAdded;

        private const int SizeRowHeight = 26;

        // Standard S/M/L/XL progression by position, regardless of each product's actual
        // SizeName text (e.g. "Regular (12oz)") — simpler and more scannable at a glance
        // than deriving a letter from the name, and matches how a real POS keypad reads.
        private static readonly string[] SizeLabels = { "S", "M", "L", "XL" };

        public ProductCardControl(Product product)
        {
            Product = product;
            Size = new Size(150, 156);
            Margin = new Padding(8);
            BackColor = BrandPalette.SurfacePrimary;
            Cursor = Cursors.Hand;

            SetStyle(ControlStyles.UserPaint | ControlStyles.AllPaintingInWmPaint | ControlStyles.OptimizedDoubleBuffer, true);

            Click += (s, e) => CardClicked?.Invoke(this, EventArgs.Empty);
            MouseEnter += (s, e) => { BackColor = BrandPalette.Tint90; Invalidate(); };
            MouseLeave += (s, e) => { BackColor = BrandPalette.SurfacePrimary; Invalidate(); };

            BuildSizeRow();
        }

        private void BuildSizeRow()
        {
            var sizes = Product.ProductSizes.ToList();
            if (sizes.Count == 0)
                return;

            var sizeRow = new Panel { Dock = DockStyle.Bottom, Height = SizeRowHeight, BackColor = BrandPalette.Brand };
            var buttonWidth = Width / sizes.Count;

            for (int i = 0; i < sizes.Count; i++)
            {
                var size = sizes[i];
                var btn = new Button
                {
                    Text = i < SizeLabels.Length ? SizeLabels[i] : (i + 1).ToString(),
                    Location = new Point(i * buttonWidth, 0),
                    Size = new Size(i == sizes.Count - 1 ? Width - i * buttonWidth : buttonWidth, SizeRowHeight),
                    FlatStyle = FlatStyle.Flat,
                    BackColor = BrandPalette.Brand,
                    ForeColor = BrandPalette.White,
                    Font = BrandPalette.BrandFontBold(8f),
                    Cursor = Cursors.Hand,
                };
                btn.FlatAppearance.BorderSize = 0;
                btn.FlatAppearance.BorderColor = BrandPalette.White;
                btn.Click += (s, e) => SizeQuickAdded?.Invoke(this, size);
                sizeRow.Controls.Add(btn);
            }

            Controls.Add(sizeRow);
        }

        protected override void OnPaint(PaintEventArgs e)
        {
            base.OnPaint(e);
            var g = e.Graphics;
            g.SmoothingMode = System.Drawing.Drawing2D.SmoothingMode.AntiAlias;

            using var borderPen = new Pen(BrandPalette.Border);
            g.DrawRectangle(borderPen, 0, 0, Width - 1, Height - 1);

            // Placeholder "image" band — a soft brand tint block where a product photo would go.
            var imageRect = new Rectangle(8, 8, Width - 16, 64);
            using (var imgBrush = new SolidBrush(BrandPalette.Tint75))
                g.FillRectangle(imgBrush, imageRect);

            using var iconFont = BrandPalette.BrandFontBold(16f);
            var iconText = "☕";

using System.ComponentModel;
using CRM_Morcaffein.Theme;

namespace CRM_Morcaffein.winforms.Views.Controls
{
    /// <summary>Simple 1-5 click-to-rate star control. No hover preview yet — click sets the value directly.</summary>
    public partial class StarRatingControl : UserControl
    {
        private int _rating;

        [DesignerSerializationVisibility(DesignerSerializationVisibility.Hidden)]
        public int Rating
        {
            get => _rating;
            set
            {
                var clamped = Math.Clamp(value, 0, 5);
                if (_rating == clamped)
                    return;

                _rating = clamped;
                Redraw();
                RatingChanged?.Invoke(this, EventArgs.Empty);
            }
        }

        public event EventHandler? RatingChanged;

        public StarRatingControl()
        {
            InitializeComponent();

            foreach (var star in _stars)
            {
                star.Click += (s, e) => Rating = (int)((Label)s!).Tag!;
                star.MouseEnter += (s, e) => Preview((int)((Label)s!).Tag!);
                star.MouseLeave += (s, e) => Redraw();
            }
        }

        private void Preview(int hoverValue)
        {
            for (int i = 0; i < _stars.Length; i++)
            {
                var filled = i < hoverValue;
                _stars[i].Text = filled ? "★" : "☆";
                _stars[i].ForeColor = filled ? BrandPalette.Brand : BrandPalette.RatingInactive;
            }
        }

        private void Redraw() => Preview(_rating);
    }

        private void RunCustomerSearch()
        {
            var query = _customerSearchTxtBx.Text.Trim();
            if (string.IsNullOrEmpty(query))
                return;

            var match = _customerService.FindByPhoneOrId(query);
            if (match is null)
            {
                _customerSearchErrorLabel.Text = $"No customer found for \"{query}\".";
                _customerProfilePanel.Visible = false;
                return;
            }

            _customerSearchErrorLabel.Text = string.Empty;
            ShowCustomerProfile(match);
            SelectCustomer(match);
        }

        private void SelectCustomer(Customer customer)
        {
            var employeeId = EmployeeSession.Current?.EmployeeId
                ?? throw new InvalidOperationException("No employee session — the Transaction view requires a logged-in employee.");

            // Search-and-Go = start a fresh transaction. If the existing cart has items, confirm
            // first — a staff member mid-sale who mistypes a customer ID shouldn't lose their work
            // silently. Empty carts (or no Controller yet) just get replaced without asking.
            if (_controller is not null && _controller.CartItems.Count > 0)
            {
                var currentCustomerName = _controller.Customer.FullName;
                var confirm = MessageBox.Show(
                    FindForm(),
                    $"Discard the in-progress order for {currentCustomerName} and start a new one for {customer.FullName}?",
                    "Start New Transaction",
                    MessageBoxButtons.YesNo,
                    MessageBoxIcon.Warning);

                if (confirm != DialogResult.Yes)
                {
                    _customerSearchTxtBx.Text = _controller.Customer.CustomerId.ToString();
                    return;
                }
            }

            _controller = new TransactionController(_orderService, _catalogService, _campaignService, _customerService, customer, employeeId);

            RefreshProductsPanel();
            RefreshCartLines();
            RefreshSummary();
        }

        /// <summary>
        /// Opens the "choose a promo" menu above the promo button. Menu items apply a promo on
        /// click; a "Clear applied promo" item appears when one is currently applied. The button
        /// label is refreshed from RefreshPromoButton() after any change, so the button always
        /// reflects the current state.
        /// </summary>
        private void ShowPromoMenu()
        {
            if (_controller is null)
                return;

            var menu = new ContextMenuStrip();

            if (_controller.Order.CampaignRedeems.Count > 0)
            {
                var clear = menu.Items.Add("Clear applied promo");
                clear.Font = BrandPalette.BrandFontBold(9f);
                clear.Click += (s, e) =>
                {
                    _controller.ClearPromos();
                    RefreshSummary();
                    RefreshPromoButton();
                };
                menu.Items.Add(new ToolStripSeparator());
            }

            var eligiblePromos = _controller.GetEligiblePromos();
            if (eligiblePromos.Count == 0)
            {
                var none = menu.Items.Add("No promos available for this order");
                none.Enabled = false;
            }
            else
            {
                foreach (var campaign in eligiblePromos)
                {
                    var alreadyApplied = _controller.Order.CampaignRedeems
                        .Any(r => r.CampaignId == campaign.CampaignId);

                    var label = campaign.DiscountType == "Percentage"
                        ? $"{campaign.PromoName} — {campaign.DiscountValue:0.##}% off"
                        : $"{campaign.PromoName} — ₱{campaign.DiscountValue:0.##} off";
                    if (alreadyApplied)
                        label += "  (applied)";

                    var item = menu.Items.Add(label);
                    item.Enabled = !alreadyApplied;
                    item.Click += (s, e) =>
                    {
                        try
                        {
                            _controller.ApplyPromo(campaign);
                            RefreshSummary();
                            RefreshPromoButton();
                        }
                        catch (Exception ex)
                        {
                            ShowCartStatus(ex.Message);
                        }
                    };
                }
            }

            // The button sits at the bottom of the column, so the menu opens above it — dropping
            // it downward would fall off the window.
            menu.Show(_promoMenuBtn, new Point(0, -menu.PreferredSize.Height));
        }

        /// <summary>
        /// Syncs the promo button's label with the currently-applied promo (if any) and
        /// enables/disables it based on whether there's an active transaction. Called from
        /// RefreshSummary (the funnel for all cart-state changes) and from ShowPromoMenu after
        /// apply/clear so the label updates immediately.
        /// </summary>
        private void RefreshPromoButton()
        {
            if (_controller is null)
            {
                _promoMenuBtn.Text = "Select a Promo ▾";
                _promoMenuBtn.Enabled = false;
                return;
            }

            var applied = _controller.Order.CampaignRedeems.FirstOrDefault();

            if (applied is null)
            {
                _promoMenuBtn.Text = "Select a Promo ▾";
            }
            else
            {
                // Look up the campaign's display name from the eligible list first; if it's
                // since become ineligible, fall back to the campaign service so the label
                // still reads right.
                var campaign = _controller.GetEligiblePromos()
                                   .FirstOrDefault(c => c.CampaignId == applied.CampaignId)
                               ?? _campaignService.GetById(applied.CampaignId);

                var name = campaign?.PromoName ?? $"Promo #{applied.CampaignId}";
                _promoMenuBtn.Text = $"{name} ▾";
            }

            _promoMenuBtn.Enabled = true;
        }

        private void RefreshSummary()
        {
            RefreshPromoButton();

            if (_controller is null)
            {
                _subTotalValueLabel.Text = "₱0.00";
                _discountValueLabel.Text = "₱0.00";
                _vatValueLabel.Text = "₱0.00";
                _pwdSeniorValueLabel.Text = "₱0.00";
                _totalValueLabel.Text = "₱0.00";
                _logSaleBtn.Enabled = false;
                return;
            }

            _subTotalValueLabel.Text = $"₱{_controller.SubTotal:0.00}";
            _discountValueLabel.Text = $"₱{_controller.TotalDiscount:0.00}";
            _vatValueLabel.Text = $"₱{_controller.TotalAmount * VatRate:0.00}";
            _pwdSeniorValueLabel.Text = "₱0.00";
            _totalValueLabel.Text = $"₱{_controller.TotalAmount:0.00}";
            _logSaleBtn.Enabled = _controller.HasItems;
        }

        private void QuickAddToCart(Product product, ProductSize size)
        {
            if (_controller is null)
            {
                ShowCartStatus("Pick a customer on the left first.");
                return;
            }

            try
            {
                _controller.AddItemToCart(product, size, 1, Enumerable.Empty<ProductModifier>());
                RefreshCartLines();
                RefreshSummary();
            }
            catch (Exception ex)
            {
                ShowCartStatus(ex.Message);
            }
        }

        private void OpenCustomizerForNewItem(Product product)
        {
            if (_controller is null)
            {
                ShowCartStatus("Pick a customer on the left first.");
                return;
            }

            using var form = new FrmItemCustomizer(product, _controller.GetModifiers());

            if (form.ShowDialog(FindForm()) == DialogResult.OK)
            {
                try
                {
                    _controller.AddItemToCart(product, form.SelectedSize, form.Quantity, form.SelectedModifiers);
                    RefreshCartLines();
                    RefreshSummary();
                }
                catch (Exception ex)
                {
                    ShowCartStatus(ex.Message);
                }
            }
        }

        private void OpenCustomizerForEdit(OrderItem item)
        {
            if (_controller is null || item.Product is null) return;

            using var form = new FrmItemCustomizer(item.Product, _controller.GetModifiers(), item);

            if (form.ShowDialog(FindForm()) == DialogResult.OK)
            {
                try
                {
                    _controller.UpdateCartItem(item, form.SelectedSize, form.Quantity, form.SelectedModifiers);
                    RefreshCartLines();
                    RefreshSummary();
                }
                catch (Exception ex)
                {
                    ShowCartStatus(ex.Message);
                }
            }
        }

        private void RefreshCartLines()
        {
            _cartLinesPanel.Controls.Clear();

            if (_controller is null || _controller.CartItems.Count == 0)
            {
                _cartLinesPanel.Controls.Add(_cartEmptyLabel);
                return;
            }

            var lineWidth = Math.Max(200,
                _cartLinesPanel.ClientSize.Width - SystemInformation.VerticalScrollBarWidth - 8);

            foreach (var item in _controller.CartItems)
            {
                var line = new CartLineControl(item) { Width = lineWidth };
                line.QuantityIncreased += (s, e) => ChangeQuantity(item, +1);
                line.QuantityDecreased += (s, e) => ChangeQuantity(item, -1);
                line.RemoveClicked += (s, e) => RemoveFromCart(item);
                _cartLinesPanel.Controls.Add(line);
            }
        }

        private void RemoveFromCart(OrderItem item)
        {
            if (_controller is null) return;

            _controller.RemoveItem(item);
            RefreshCartLines();
            RefreshSummary();
        }

        /// <summary>
        /// The Order box's inline "-"/"+" stepper. TransactionController has no dedicated
        /// "just change the quantity" method — UpdateCartItem needs the real ProductSize and
        /// ProductModifier domain objects, not just the OrderItem's snapshotted values, so
        /// they're looked back up from the catalog here. Dropping to 0 removes the line.
        /// </summary>
        private void ChangeQuantity(OrderItem item, int delta)
        {
            if (_controller is null || item.Product is null) return;

            var newQuantity = item.Quantity + delta;
            if (newQuantity <= 0)
            {
                RemoveFromCart(item);
                return;
            }

            var size = item.Product.ProductSizes.First(s => s.ProductSizeId == item.ProductSizeId);
            var modifiers = _controller.GetModifiers()
                .Where(m => item.OrderItemModifiers.Any(om => om.ProductModifierId == m.ProductModifierId));

            try
            {
                _controller.UpdateCartItem(item, size, newQuantity, modifiers);
                RefreshCartLines();
                RefreshSummary();
            }
            catch (Exception ex)
            {
                ShowCartStatus(ex.Message);
            }
        }

        private void ShowCartStatus(string message) => _cartStatusLabel.Text = message;

        private void CompleteSale()
        {
            if (_controller is null) return;

            try
            {
                var customerName = _controller.Customer.FullName;
                var order = _controller.CompleteSale();

                MessageBox.Show(FindForm(),
                    $"Sale completed. {order.PointsEarned} points awarded to {customerName}.",
                    "Sale Complete", MessageBoxButtons.OK, MessageBoxIcon.Information);

                // Ready for the next customer — a fresh Controller/Order is created the next
                // time someone is found via the search box.
                _controller = null;
                _customerProfilePanel.Visible = false;
                _customerSearchErrorLabel.Text = string.Empty;
                _customerSearchTxtBx.Text = string.Empty;
                RefreshProductsPanel();
                RefreshCartLines();
                RefreshSummary();
            }
            catch (Exception ex)
            {
                ShowCartStatus(ex.Message);
            }
        }
    }
}
